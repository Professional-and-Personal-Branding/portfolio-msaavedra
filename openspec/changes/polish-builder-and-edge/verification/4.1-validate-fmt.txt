  # Caddy sirve la SPA y hace de proxy hacia la API.
  #
  # Dos razones para el proxy, no solo comodidad:
  #   1. Un único origen → la cookie de sesión puede ir con SameSite=strict
  #      (la fila más segura de la tabla de docs/DEPLOY.md § 0) y no hay CORS.
  #   2. /robots.txt y /sitemap.xml los genera el backend, y así quedan
  #      en el dominio del sitio, que es donde los buscadores los esperan.
  #
  # Variables que inyecta Railway:
  #   PORT        el puerto en el que tenemos que escuchar.
  #   API_ORIGIN  la API por la red privada, p. ej. http://api.railway.internal:8080
  #
  # Railway termina el TLS en su borde, así que acá no hay certificados que gestionar.
  {
  	admin off
  	auto_https off
  }
  
  :{$PORT:8080} {
  	encode zstd gzip
  
  	# Fragmento de head por perfil (change `add-server-rendered-head`). Lo pide el
  	# propio Caddy desde `index.html` con `httpInclude`, que hace una sub-petición a
  	# este mismo servidor: por eso es una ruta y no `{$API_ORIGIN}` directo. Va
  	# antes que @backend y con su propio presupuesto, para que una API lenta no
  	# demore el HTML más de medio segundo por fase.
  	@head path /api/public/profile-head
  	handle @head {
  		reverse_proxy {$API_ORIGIN} {
  			header_up X-Forwarded-Proto https
  			transport http {
  				dial_timeout 500ms
  				response_header_timeout 500ms
  			}
  			# Un 4xx o 5xx de la API (la base caída, por ejemplo) no es un error de
  			# Caddy: se reenviaría tal cual, `httpInclude` fallaría por el estado y
  			# con él todo el HTML. Se cambia por el mismo fragmento vacío con 200 que
  			# deja `handle_errors` cuando la API no responde.
  			@apiFailed status 4xx 5xx
  			handle_response @apiFailed {
  				respond "" 200
  			}
  		}
  	}
  
  	# `/health*` también va al proxy: si no, el fallback de la SPA responde 200 con
  	# HTML y la verificación del § 5 del runbook pasaría sin haber tocado la API.
  	@backend path /api/* /health /health/live /robots.txt /sitemap.xml
  	handle @backend {
  		reverse_proxy {$API_ORIGIN} {
  			# Explícito, para que la redirección a https del backend no entre en bucle:
  			# el TLS real lo terminó el borde de Railway antes de llegar hasta acá.
  			header_up X-Forwarded-Proto https
  		}
  	}
  
  	handle {
  		root * /srv
  		# El fallback de la SPA: sin esto, /tech-lead-dotnet daría 404.
  		try_files {path} /index.html
  		# Solo `index.html` pasa por templates, con delimitadores que en el archivo
  		# son comentarios HTML: así sigue siendo válido para Vite y el navegador.
  		@index path /index.html
  		templates @index {
  			mime text/html
  			between <!--# -->
  		}
  		file_server
  
  		# CSP del sitio (change `polish-builder-and-edge`, D7). Solo en este bloque:
  		# lo que pasa por el proxy (@backend, @head) no la recibe desde el borde.
  		# - script-src 'self' sin 'unsafe-inline': el build no deja scripts inline. El
  		#   JSON-LD es un data block (type="application/ld+json"): no se ejecuta y
  		#   script-src no se le aplica, así que no necesita hash ni nonce.
  		# - style-src 'unsafe-inline': `sonner` inserta un <style> y no admite nonce
  		#   (el riesgo residual es inyección de CSS, no de script); más la hoja de
  		#   Google Fonts, cuyos archivos salen de fonts.gstatic.com.
  		# - Sin upgrade-insecure-requests: Railway termina el TLS en el borde y, en
  		#   local por http, esa directiva rompería la carga de los propios assets.
  		# El nombre de la cabecera sale de CSP_HEADER: con
  		# `Content-Security-Policy-Report-Only` la política solo reporta, sin rebuild.
  		header {$CSP_HEADER:Content-Security-Policy} "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data:; connect-src 'self'; object-src 'none'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'"
  	}
  
  	# Degradación: si la API no responde a tiempo o rechaza la conexión, el
  	# fragmento llega vacío con 200 y `index.html` entrega su head de fallback (si
  	# responde con error, lo cubre `handle_response` arriba). Un error acá haría
  	# fallar el template y, con él, todo el HTML: el sitio no depende de la API.
  	handle_errors {
  		@headError path /api/public/profile-head
  		handle @headError {
  			respond "" 200
  		}
  	}

  	}

  		output stdout
  		format json
  	}
